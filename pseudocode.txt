PROGRAM FreetownCommuterTransitSystem

// ---------- CONSTANTS ----------
CONSTANT ZONE1_FARE      = 5.00     // short distance (NLe)
CONSTANT ZONE2_FARE      = 10.00    // medium distance (NLe)
CONSTANT ZONE3_FARE      = 15.00    // long distance (NLe)
CONSTANT PEAK_MULTIPLIER = 1.25     // fixed 25% peak-hour surcharge

// ---------- VARIABLES ----------
DECLARE routeID                                  AS String
DECLARE maxCapacity, passengerCount, distanceZone AS Integer
DECLARE peakHour, boardMore                      AS String
DECLARE finalFare                                AS Real

// ---------- FUNCTIONS ----------
FUNCTION GetValidCapacity() RETURNS Integer
    DISPLAY "Enter vehicle maximum capacity:"
    INPUT capacity
    WHILE capacity <= 0 DO
        DISPLAY "INVALID CAPACITY! Enter a number greater than 0:"
        INPUT capacity
    END WHILE
    RETURN capacity
END FUNCTION

FUNCTION GetValidCount(maxSeats) RETURNS Integer
    DISPLAY "Enter initial passenger count:"
    INPUT count
    WHILE count < 0 OR count > maxSeats DO
        DISPLAY "INVALID COUNT! Enter a number from 0 to ", maxSeats, ":"
        INPUT count
    END WHILE
    RETURN count
END FUNCTION

FUNCTION GetValidZone() RETURNS Integer
    DISPLAY "Enter zone (1-3):"
    INPUT zone
    WHILE zone < 1 OR zone > 3 DO
        DISPLAY "INVALID ZONE! Please enter a zone between 1 and 3:"
        INPUT zone
    END WHILE
    RETURN zone
END FUNCTION

FUNCTION GetYesNo(question) RETURNS String
    DISPLAY question
    INPUT answer
    SET answer = UPPERCASE(answer)
    WHILE answer <> "YES" AND answer <> "NO" DO
        DISPLAY "INVALID ENTRY! Type YES or NO:"
        INPUT answer
        SET answer = UPPERCASE(answer)
    END WHILE
    RETURN answer
END FUNCTION

FUNCTION CalculateFare(zone, isPeak) RETURNS Real
    IF zone = 1 THEN
        SET baseFare = ZONE1_FARE
    ELSE IF zone = 2 THEN
        SET baseFare = ZONE2_FARE
    ELSE
        SET baseFare = ZONE3_FARE
    END IF
    IF isPeak = "YES" THEN
        SET fare = baseFare * PEAK_MULTIPLIER
    ELSE
        SET fare = baseFare
    END IF
    RETURN fare
END FUNCTION

MODULE DisplayReceipt(route, zone, fare, count, maxSeats)
    DISPLAY "BOARDING SUCCESSFUL!"
    DISPLAY "Route: ", route, "  Zone: ", zone
    DISPLAY "Fare: NLe ", fare
    DISPLAY "Passengers on board: ", count, "/", maxSeats
END MODULE

// ---------- MAIN PROGRAM ----------
MAIN
    DISPLAY "=== FREETOWN COMMUTER TRANSIT SYSTEM ==="
    DISPLAY "Enter Route ID:"
    INPUT routeID
    SET maxCapacity    = GetValidCapacity()
    SET passengerCount = GetValidCount(maxCapacity)
    SET boardMore      = "YES"

    IF passengerCount = maxCapacity THEN
        DISPLAY "VEHICLE IS FULL! No boarding possible."
        SET boardMore = "NO"
    END IF

    WHILE boardMore = "YES" AND passengerCount < maxCapacity DO
        SET distanceZone = GetValidZone()
        SET peakHour     = GetYesNo("Is it peak hour? (YES/NO):")
        SET finalFare    = CalculateFare(distanceZone, peakHour)
        SET passengerCount = passengerCount + 1
        CALL DisplayReceipt(routeID, distanceZone, finalFare, passengerCount, maxCapacity)

        IF passengerCount = maxCapacity THEN
            DISPLAY "VEHICLE IS FULL!"
            SET boardMore = "NO"
        ELSE
            SET boardMore = GetYesNo("Board more passengers? (YES/NO):")
        END IF
    END WHILE

    DISPLAY "Session finished. Have a safe journey!"
END MAIN
